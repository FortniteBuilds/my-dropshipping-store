PayPal Integration Code 
// Secure PayPal payment processing
// Using environment variables for credentials
const paypal = require('@paypal/checkout-server-sdk');

const Environment = process.env.PAYPAL_ENV === 'production' 
  ? paypal.core.LiveEnvironment 
  : paypal.core.SandboxEnvironment;

const paypalClient = new paypal.core.PayPalHttpClient(
  new Environment(
    process.env.PAYPAL_CLIENT_ID,
    process.env.PAYPAL_SECRET
  )
);

async function createOrder(amount) {
  const request = new paypal.orders.OrdersCreateRequest();
  request.prefer("return=representation");
  request.requestBody({
    intent: 'CAPTURE',
    purchase_units: [{
      amount: {
        currency_code: 'USD',
        value: amount
      }
    }]
  });
  
  const response = await paypalClient.execute(request);
  return response.result.id;
}

Discord Webhook Integration 
// Send order notification to your private Discord
// Webhook URL loaded from environment variable
const sendDiscordNotification = (order) => {
  const webhookURL = process.env.DISCORD_WEBHOOK_URL;
  
  const embed = {
    title: `New Order #${order.id}`,
    description: `**Customer:** ${order.customer.name}\n**Total:** $${order.total}`,
    color: 0x8a2be2,
    fields: [
      { name: 'Items', value: order.items.join('\n') },
      { name: 'Status', value: 'Processing' }
    ]
  };

  fetch(webhookURL, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ embeds: [embed] })
  }).catch(err => console.error('Discord notification failed'));
};

Enable HTTPS

// Express.js HTTPS enforcement
app.use((req, res, next) => {
  if (req.headers['x-forwarded-proto'] !== 'https') {
    return res.redirect('https://' + req.headers.host + req.url);
  }
  next();
});
